! Title: X (Twitter) reading tweaks for Brave
! Description: Bigger tweet text, bigger display names, a full-width action row,
!   a home feed with the avatar beside the name and everything else full width,
!   and a decluttered menu (Grok, Premium, Calls, Ads, Creator Studio,
!   Business, Lists).
! Homepage: https://github.com/urlittledollstacy-web/x-tweaks
! Source userstyle: https://userstyles.world/style/31054 (X test 2)
! Expires: 1 days
! Version: 1.2.0
!
! HOW TO USE
!   1. Open brave://settings/shields/filters
!   2. Add filter list by URL:
!      https://raw.githubusercontent.com/urlittledollstacy-web/x-tweaks/main/x-tweaks-brave.txt
!   3. Reload x.com.
!
! This file is pure ASCII on purpose, and Expires is one day so updates arrive.
!
! 1. BIGGER TEXT
!    Was: div[data-testid="tweetText"] { font-size: 17px; line-height: 1.45; }
!    X sets 15px/20px. The text container is the dir="auto" block that wraps it.
x.com##article div[dir="auto"][class*="whitespace-pre-wrap"]:style(font-size: 17px !important; line-height: 1.45 !important;)
x.com##article div[class*="whitespace-pre-wrap"] span:style(font-size: inherit !important; line-height: inherit !important;)
!    Was: [data-testid="User-Name"] { font-size: 16px; }
!    Only the bold display name, not the grey handle.
x.com##article a[href] div[class*="font-bold"]:style(font-size: 16px !important; line-height: 1.35 !important;)
!
! 2. ACTION ROW
!    Was: article [role="group"] { display: flex; justify-content: space-between; }
!    The current action row is the div holding the reply/repost/like/bookmark
!    buttons. X lays them out as flex: 1 1 0%, so all four are 115px wide and
!    each icon floats in empty space. flex: 0 0 auto makes them hug their icons.
x.com##article div:has(> [data-engagement-action]):style(display: flex !important; justify-content: space-between !important; align-items: center !important; padding-block: 10px !important; min-height: 52px !important;)
x.com##article div:has(> [data-engagement-action]) > div:style(flex: 0 0 auto !important; width: auto !important; min-width: 0 !important;)
x.com##article div:has(> [data-engagement-action]) svg:style(width: 20px !important; height: 20px !important;)
x.com##article div:has(> [data-engagement-action]) [class*="tabular-nums"]:style(font-size: 13px !important; white-space: nowrap !important;)
!
! 3. HOME FEED LAYOUT
!    Was, scoped with @-moz-document url-prefix("https://x.com/home"):
!      article div:has(> div.w-10) > div > :first-child { ... }
!    The point: the avatar and the name/handle share the top row, and everything
!    else is full width.
!    Two changes were needed for the current markup. X nests the name row one
!    level deeper, and for short posts the tweet text lives INSIDE the name row,
!    so moving the name row into column 2 drags the text with it. The name row
!    is dissolved too, so only the name/handle block takes column 2 and the text
!    spans the full width.
x.com##article div:matches-path(/home/):has(> div[class*="w-10"] > div > a > div[class*="x-avatar"]):style(display: grid !important; grid-template-columns: 40px minmax(0, 1fr) !important; column-gap: 8px !important; row-gap: 4px !important; align-items: start !important;)
x.com##article div:matches-path(/home/):has(> div[class*="w-10"] > div > a > div[class*="x-avatar"]) > div[class*="w-10"]:style(grid-column: 1 !important; grid-row: 1 !important; align-self: center !important;)
x.com##article div:matches-path(/home/):has(> div[class*="w-10"] > div > a > div[class*="x-avatar"]) > div[class*="flex-1"]:style(display: contents !important;)
x.com##article div:matches-path(/home/):has(> div[class*="w-10"] > div > a > div[class*="x-avatar"]) > div[class*="flex-1"] > div[class*="gap-0.5"]:style(display: contents !important;)
x.com##article div:matches-path(/home/):has(> div[class*="w-10"] > div > a > div[class*="x-avatar"]) div[class*="@container/post-actions"]:style(grid-column: 2 !important; grid-row: 1 !important; align-self: center !important; min-width: 0 !important;)
x.com##article div:matches-path(/home/):has(> div[class*="w-10"] > div > a > div[class*="x-avatar"]) > div[class*="flex-1"] > div[class*="gap-0.5"] > div[dir="auto"]:style(grid-column: 1 / -1 !important; grid-row: 2 !important; min-width: 0 !important;)
x.com##article div:matches-path(/home/):has(> div[class*="w-10"] > div > a > div[class*="x-avatar"]) > div[class*="flex-1"] > :not(div[class*="gap-0.5"]):style(grid-column: 1 / -1 !important; min-width: 0 !important;)
!
! 4. DECLUTTER
!    Every route rule is anchored to a relative href, so an external link such
!    as https://example.com/i/lists is never hidden.
!    Every procedural rule ends in :style(display: none !important); a bare
!    :has-text() or :upward() is not CSS and does nothing.

x.com##button[aria-label*="Grok"]
x.com##a[href="/i/grok"]

x.com##a[aria-label="Grok"]
x.com##div[role="menuitem"]:has-text(Grok):style(display: none !important)
